-- ═══════════════════════════════════════════════════════════════════
-- TABLA 1: productos
-- ═══════════════════════════════════════════════════════════════════
CREATE TABLE IF NOT EXISTS productos (
    id          SERIAL PRIMARY KEY,
    nombre      VARCHAR(200) NOT NULL,
    categoria   VARCHAR(100),
    precio_usd  DOUBLE PRECISION DEFAULT 0.0,
    precio_ves  DOUBLE PRECISION DEFAULT 0.0,
    activo      BOOLEAN DEFAULT TRUE,
    creado_en   TIMESTAMP DEFAULT NOW()
);


-- ═══════════════════════════════════════════════════════════════════
-- TABLA 2: tasas_cambio
-- ═══════════════════════════════════════════════════════════════════
CREATE TABLE IF NOT EXISTS tasas_cambio (
    id          SERIAL PRIMARY KEY,
    fuente      VARCHAR(50) NOT NULL,
    tasa        DOUBLE PRECISION NOT NULL,
    timestamp   TIMESTAMP DEFAULT NOW()
);


-- ═══════════════════════════════════════════════════════════════════
-- TABLA 3: log_cambios
-- ═══════════════════════════════════════════════════════════════════
CREATE TABLE IF NOT EXISTS log_cambios (
    id                SERIAL PRIMARY KEY,
    producto_id       INTEGER,
    producto_nombre   VARCHAR(200),
    precio_antes      DOUBLE PRECISION,
    precio_despues    DOUBLE PRECISION,
    variacion_pct     DOUBLE PRECISION,
    tasa_usada        DOUBLE PRECISION,
    aprobado          BOOLEAN,
    timestamp         TIMESTAMP DEFAULT NOW()
);


-- ═══════════════════════════════════════════════════════════════════
-- ÍNDICES (para que las búsquedas sean rápidas)
-- ═══════════════════════════════════════════════════════════════════
CREATE INDEX IF NOT EXISTS idx_productos_activo
    ON productos(activo);

CREATE INDEX IF NOT EXISTS idx_tasas_fuente_fecha
    ON tasas_cambio(fuente, timestamp DESC);

CREATE INDEX IF NOT EXISTS idx_log_producto
    ON log_cambios(producto_id);
